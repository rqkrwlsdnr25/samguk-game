  for(const entry of TERRITORIES){
    if(world.territories.some(t=>String(t?.id||'')===entry.id||String(t?.name||'')===entry.name))continue;
    world.territories.push({
      id:entry.id,name:entry.name,x:entry.x,y:entry.y,home:entry.home,path:entry.path,
      resources:Array.isArray(entry.resources)?entry.resources.slice():[],terrainType:entry.terrainType||null,mapExpansion:'v141'
    });
    world.neighbors.push([]);
  }

  // SVG/Canvas land silhouette uses the corrected base polygons + expansion polygons only.
  // This removes the old blue coastal wedge physically instead of painting sea over stale land.
  world.baseLandPath=BASE_LAND_PATH;
  world.landPath=BASE_LAND_PATH+TERRITORIES.map(t=>t.path).join('');

  addNamedLinks(LAND_LINKS,false);
  addNamedLinks(SEA_LINKS,true);

  const seaSet=new Set((world.seaRoutes||[]).map(pair=>routeKey(pair[0],pair[1])));
  for(let i=0;i<world.territories.length;i++){
    const territory=world.territories[i];
    const connected=Array.isArray(world.neighbors[i])?world.neighbors[i].slice().sort((a,b)=>a-b):[];
    territory.connectedRoutes=connected;
    territory.connectedSeaRoutes=connected.filter(j=>seaSet.has(routeKey(i,j)));
  }

  world.minX=Math.min(Number.isFinite(Number(world.minX))?Number(world.minX):-180.5,-180.5);
  world.minY=Math.min(Number.isFinite(Number(world.minY))?Number(world.minY):-.5,-.5);

  world.height=Math.max(Number(world.height)||0,1271.0);
  world.expandedBounds=Object.freeze({minX:-180.5,minY:-.5,maxX:1120.75,maxY:1270.5,width:1301.25,height:1271.0});
  world.__mapExpansionV125Applied=true;

  function territoryId(name){return nameMap().get(String(name))}
  function canMoveSeaRoute(a,b){
    const x=Number(a),y=Number(b);
    if(!Number.isInteger(x)||!Number.isInteger(y)||x===y)return false;
    if(!world.territories[x]||!world.territories[y])return false;
    return seaSet.has(routeKey(x,y));
  }
  function connectedRoutes(id){
    const i=Number(id);
    return Number.isInteger(i)&&Array.isArray(world.neighbors[i])?world.neighbors[i].slice():[];
  }
  function appendTerrainRules(target){
    if(!target||!Array.isArray(target.areas))return false;
    for(let i=0;i<TERRAIN_RULES.length;i++){
      const targetIndex=START_INDEX+i;
      if(!target.areas[targetIndex])target.areas[targetIndex]={...TERRAIN_RULES[i]};
    }
    while(target.areas.length<world.territories.length)target.areas.push({kind:'plain',river:false});
    return true;
  }

    version:141,startIndex:START_INDEX,addedCount:world.territories.length-START_INDEX,
