  function ensureNeighborSlot(){while(world.neighbors.length<world.territories.length)world.neighbors.push([])}
  function addNeighbor(a,b){
    if(!Number.isInteger(a)||!Number.isInteger(b)||a<0||b<0||a===b)return false;
    ensureNeighborSlot();
    if(!world.neighbors[a].includes(b))world.neighbors[a].push(b);
    if(!world.neighbors[b].includes(a))world.neighbors[b].push(a);
    world.neighbors[a].sort((x,y)=>x-y);
    world.neighbors[b].sort((x,y)=>x-y);
    return true;
  }
  function addSeaRoute(a,b){
    if(!Array.isArray(world.seaRoutes))world.seaRoutes=[];
    const key=routeKey(a,b);
    if(!world.seaRoutes.some(pair=>Array.isArray(pair)&&pair.length>=2&&routeKey(pair[0],pair[1])===key)){
      world.seaRoutes.push([Math.min(a,b),Math.max(a,b)]);
    }
    return addNeighbor(a,b);
  }
  function addNamedLinks(links,sea=false){
    const ids=nameMap();
    for(const [from,to] of links){
      const a=ids.get(from),b=ids.get(to);
      if(!Number.isInteger(a)||!Number.isInteger(b))continue;
      if(sea)addSeaRoute(a,b); else addNeighbor(a,b);
    }
  }

      if(!target.areas[targetIndex])target.areas[targetIndex]={...TERRAIN_RULES[i]};
    }
    while(target.areas.length<world.territories.length)target.areas.push({kind:'plain',river:false});
    return true;
